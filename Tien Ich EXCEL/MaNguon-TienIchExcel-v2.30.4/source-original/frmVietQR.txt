Private Sub btnCreate_Click()
    Dim bankSelect As String, bankId As String
    Dim accNo As String, amount As String, memo As String
    Dim imgURL As String, selRange As Range, targetRng As Range
    Dim shp As Shape, ws As Worksheet
    Dim picSize As Double
    
    If cboBank.ListIndex = -1 Then
        MsgBox "Vui long chon mot ngan hang!", vbExclamation, "Thong bao"
        cboBank.SetFocus
        Exit Sub
    End If
    
    bankSelect = Trim(cboBank.Value)
    bankId = Trim(Split(bankSelect, "-")(0))
    
    accNo = Trim(txtAcc.Text)
    If accNo = "" Then
        MsgBox "Vui long nhap so tai khoan!", vbExclamation, "Thong bao"
        txtAcc.SetFocus
        Exit Sub
    End If
    
    ' --- LUU LAI NGAN HANG & SO TAI KHOAN LAM MAC DINH CHO LAN SAU ---
    SaveSetting "ExcelAddinBarcode", "VietQRConfig", "DefaultBankIndex", CStr(cboBank.ListIndex)
    SaveSetting "ExcelAddinBarcode", "VietQRConfig", "DefaultAccNo", accNo
    
    amount = Trim(txtAmount.Text)
    memo = Trim(txtMemo.Text)
    picSize = 120
    
    If TypeName(Selection) <> "Range" Then
        MsgBox "Vui long chon o tinh de dat ma QR!", vbExclamation, "Thong bao"
        Exit Sub
    End If
    
    Set selRange = Selection.Cells(1, 1)
    Set ws = selRange.Worksheet
    Set targetRng = selRange.Offset(0, 1)
    
    If selRange.EntireRow.RowHeight < 130 Then selRange.EntireRow.RowHeight = 130
    If targetRng.EntireColumn.ColumnWidth < 25 Then targetRng.EntireColumn.ColumnWidth = 25
    
    For Each shp In ws.Shapes
        If shp.Type = msoPicture Or shp.Type = msoLinkedPicture Then
            If Not Intersect(shp.TopLeftCell, targetRng) Is Nothing Then shp.Delete
        End If
    Next shp
    
    imgURL = "https://img.vietqr.io/image/" & bankId & "-" & accNo & "-compact2.png"
    If amount <> "" Or memo <> "" Then
        imgURL = imgURL & "?amount=" & Application.EncodeURL(amount) & "&addInfo=" & Application.EncodeURL(memo)
    End If
    
    On Error Resume Next
    Set shp = ws.Shapes.AddPicture( _
                Filename:=imgURL, _
                LinkToFile:=msoFalse, _
                SaveWithDocument:=msoTrue, _
                Left:=targetRng.Left + (targetRng.Width - picSize) / 2, _
                Top:=targetRng.Top + (targetRng.Height - picSize) / 2, _
                Width:=picSize, _
                Height:=picSize)
    If Not shp Is Nothing Then shp.Placement = xlMoveAndSize
    On Error GoTo 0
    
    Unload Me
    MsgBox "Da tao ma VietQR thanh cong!", vbInformation, "Hoan tat"
End Sub

Private Sub btnClose_Click()
    Unload Me
End Sub
