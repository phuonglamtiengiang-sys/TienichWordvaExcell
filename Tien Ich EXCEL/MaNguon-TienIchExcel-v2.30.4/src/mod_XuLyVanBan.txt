' Ham giai ma Unicode dang \uXXXX
Private Function UVanBan(ByVal str As String) As String
    Dim i As Long, res As String, hexCode As String
    res = ""
    i = 1
    Do While i <= Len(str)
        If Mid(str, i, 2) = "\u" And i + 5 <= Len(str) Then
            hexCode = Mid(str, i + 2, 4)
            res = res & ChrW(val("&H" & hexCode))
            i = i + 6
        Else
            res = res & Mid(str, i, 1)
            i = i + 1
        End If
    Loop
    UVanBan = res
End Function

Private Sub ShowMsg(ByVal prompt As String, Optional ByVal title As String = "Th\u00f4ng b\u00e1o", Optional ByVal buttons As Long = 64)
    MessageBoxW Application.hwnd, StrPtr(UVanBan(prompt)), StrPtr(UVanBan(title)), buttons
End Sub

' Ham lam sach khoang trang dau, giua va cuoi chuoi
Public Function LamSachText(ByVal txt As Variant) As String
    Dim s As String
    If IsError(txt) Or IsNull(txt) Then Exit Function
    s = CStr(txt)
    s = Replace(s, ChrW(160), " ")
    s = Replace(s, ChrW(8203), "")
    s = Replace(s, ChrW(-257), "")
    s = Replace(s, vbTab, " ")
    s = Replace(s, vbCr, " ")
    s = Replace(s, vbLf, " ")
    
    Do While InStr(s, "  ") > 0
        s = Replace(s, "  ", " ")
    Loop
    
    LamSachText = Trim(s)
End Function

' In thuong Unicode
Public Function ToLowerUnicode(ByVal txt As String) As String
    Static sLower As String, sUpper As String
    Dim res As String
    Dim i As Long, p As Long, ch As String
    If Len(sLower) = 0 Then
        sLower = UVanBan(UNI_LOWER)
        sUpper = UVanBan(UNI_UPPER)
    End If
    res = ""
    For i = 1 To Len(txt)
        ch = Mid(txt, i, 1)
        p = InStr(1, sUpper, ch, vbBinaryCompare)
        If p > 0 Then
            res = res & Mid(sLower, p, 1)
        Else
            res = res & LCase(ch)
        End If
    Next i
    ToLowerUnicode = res
End Function

' In hoa Unicode
Public Function ToUpperUnicode(ByVal txt As String) As String
    Static sLower As String, sUpper As String
    Dim res As String
    Dim i As Long, p As Long, ch As String
    If Len(sLower) = 0 Then
        sLower = UVanBan(UNI_LOWER)
        sUpper = UVanBan(UNI_UPPER)
    End If
    res = ""
    For i = 1 To Len(txt)
        ch = Mid(txt, i, 1)
        p = InStr(1, sLower, ch, vbBinaryCompare)
        If p > 0 Then
            res = res & Mid(sUpper, p, 1)
        Else
            res = res & UCase(ch)
        End If
    Next i
    ToUpperUnicode = res
End Function

' Viet hoa dau cau (Sentence Case)
Public Function ToSentenceCase(ByVal txt As String) As String
    Dim s As String, i As Long, ch As String, newSentence As Boolean
    s = ToLowerUnicode(LamSachText(txt)): newSentence = True
    For i = 1 To Len(s)
        ch = Mid$(s, i, 1)
        If newSentence And ToUpperUnicode(ch) <> ToLowerUnicode(ch) Then
            ch = ToUpperUnicode(ch): newSentence = False
        End If
        ToSentenceCase = ToSentenceCase & ch
        If InStr(1, ".!?", ch, vbBinaryCompare) > 0 Then newSentence = True
    Next i
End Function

' Viet hoa danh tu rieng (dau moi tu)
Public Function ToTitleCase(ByVal txt As String) As String
    Dim words() As String, i As Long, w As String, res As String
    Dim cleanStr As String
    cleanStr = LamSachText(txt)
    If Len(cleanStr) = 0 Then ToTitleCase = "": Exit Function
    
    words = Split(cleanStr, " ")
    res = ""
    For i = LBound(words) To UBound(words)
        w = Trim(words(i))
        If Len(w) > 0 Then
            w = ToUpperUnicode(Left(w, 1)) & ToLowerUnicode(Mid(w, 2))
            If res = "" Then
                res = w
            Else
                res = res & " " & w
            End If
        End If
    Next i
    ToTitleCase = res
End Function

' Tach ho & dem
Public Function TachHoDem(ByVal fullName As Variant) As String
    Dim s As String, pos As Long
    s = LamSachText(fullName)
    pos = InStrRev(s, " ")
    If pos > 0 Then
        TachHoDem = Trim(Left(s, pos - 1))
    Else
        TachHoDem = ""
    End If
End Function

' Tach ten rieng
Public Function TachTen(ByVal fullName As Variant) As String
    Dim s As String, pos As Long
    s = LamSachText(fullName)
    pos = InStrRev(s, " ")
    If pos > 0 Then
        TachTen = Trim(Mid(s, pos + 1))
    Else
        TachTen = s
    End If
End Function

Private Sub RunTextAction(ByVal mode As String)
    Dim inputRange As Range, cell As Range, result As String, oldScreen As Boolean, oldEvents As Boolean
    Dim columns As Long, reason As String
    If mode = "split" Then columns = 2
    If Not SelectedInput(inputRange, columns, mode = "split") Then Exit Sub
    UndoBegin inputRange, columns, "Xu ly van ban"
    oldScreen = Application.ScreenUpdating: oldEvents = Application.EnableEvents
    On Error GoTo Failed
    Application.ScreenUpdating = False: Application.EnableEvents = False
    For Each cell In inputRange.Cells
        If Not IsError(cell.Value2) Then
            If VarType(cell.Value2) = vbString And Not cell.HasFormula Or mode = "split" Then
                Select Case mode
                    Case "split"
                        WriteText cell.Offset(0, 1), TachHoDem(cell.Value2)
                        WriteText cell.Offset(0, 2), TachTen(cell.Value2)
                    Case "accent": WriteText cell, BoDau(cell.Value2)
                    Case "upper": WriteText cell, ToUpperUnicode(cell.Value2)
                    Case "lower": WriteText cell, ToLowerUnicode(cell.Value2)
                    Case "sentence": WriteText cell, ToSentenceCase(cell.Value2)
                    Case "title": WriteText cell, ToTitleCase(cell.Value2)
                    Case "clean": WriteText cell, LamSachText(cell.Value2)
                End Select
            End If
        End If
    Next cell
Done:
    Application.ScreenUpdating = oldScreen: Application.EnableEvents = oldEvents
    If reason <> "" Then ShowMsg reason, "Loi xu ly", 48
    UndoCommit
    Exit Sub
Failed:
    reason = Err.Description: Resume Done
End Sub
