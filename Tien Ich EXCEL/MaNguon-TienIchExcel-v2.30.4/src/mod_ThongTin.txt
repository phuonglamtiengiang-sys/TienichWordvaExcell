' Ham giai ma Unicode dang \uXXXX dung chung cho toan bo Add-in
Public Function u(ByVal str As String) As String
    Dim i As Long, res As String, hexCode As String, codeUnit As Long
    res = "": i = 1
    Do While i <= Len(str)
        If Mid$(str, i, 2) = "\u" And i + 5 <= Len(str) Then
            hexCode = Mid$(str, i + 2, 4)
            If hexCode Like "[0-9A-Fa-f][0-9A-Fa-f][0-9A-Fa-f][0-9A-Fa-f]" Then
                codeUnit = CLng("&H" & hexCode)
                If codeUnit > 32767 Then codeUnit = codeUnit - 65536
                res = res & ChrW$(codeUnit)
                i = i + 6
            Else
                res = res & Mid$(str, i, 1): i = i + 1
            End If
        Else
            res = res & Mid$(str, i, 1): i = i + 1
        End If
    Loop
    u = res
End Function

Public Sub ShowMsg(ByVal prompt As String, Optional ByVal title As String = "Th\u00f4ng b\u00e1o", Optional ByVal buttons As Long = 64)
    MessageBoxW Application.hwnd, StrPtr(u(prompt)), StrPtr(u(title)), buttons
End Sub

Public Sub SetUserFormCaptionUnicode(ByVal temporaryCaption As String, ByVal unicodeCaption As String)
    On Error Resume Next
#If VBA7 Then
    Dim formHwnd As LongPtr
#Else
    Dim formHwnd As Long
#End If
    Dim searchText As String, captionText As String
    searchText = temporaryCaption
    captionText = u(unicodeCaption)
    If Len(searchText) > 0 Then formHwnd = FindWindowW(0, StrPtr(searchText))
    If formHwnd = 0 Then formHwnd = GetActiveWindow()
    If formHwnd <> 0 Then SetWindowTextW formHwnd, StrPtr(captionText)
End Sub
